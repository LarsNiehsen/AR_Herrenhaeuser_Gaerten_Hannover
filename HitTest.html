const video = document.getElementById("video"); const canvas = document.getElementById("canvas"); const renderer = new THREE.WebGLRenderer({ canvas, alpha: true }); renderer.setSize(window.innerWidth, window.innerHeight); const scene = new THREE.Scene(); // --- camera --- const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.01, 1000); scene.add(camera); // --- cylinder --- const cylinder = new THREE.Mesh( new THREE.CylinderGeometry(0.03, 0.03, 0.1, 32), new THREE.MeshNormalMaterial() ); cylinder.position.y = 0.05; // raise so it sits on marker const markerGroup = new THREE.Group(); markerGroup.add(cylinder); // --- axes helper --- const axes = new THREE.AxesHelper(0.1); // 10cm axes markerGroup.add(axes); // Hide at start markerGroup.visible = false; scene.add(markerGroup); // --- start video --- navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" } }) .then(stream => { video.srcObject = stream; video.play(); }); // --- openCV init wait loop --- let cvReady = false; cv['onRuntimeInitialized'] = () => cvReady = true; // --- main render loop --- function tick() { requestAnimationFrame(tick); if (cvReady && video.videoWidth > 0) { detectAruco(); } renderer.render(scene, camera); } tick(); // --- ArUco detection + pose --- function detectAruco() { const src = new cv.Mat(video.videoHeight, video.videoWidth, cv.CV_8UC4); const cap = new cv.VideoCapture(video); cap.read(src); const gray = new cv.Mat(); cv.cvtColor(src, gray, cv.COLOR_RGBA2GRAY); const dictionary = new cv.aruco.getPredefinedDictionary(cv.aruco.DICT_6X6_250); const markerCorners = new cv.MatVector(); const markerIds = new cv.Mat(); cv.aruco.detectMarkers(gray, dictionary, markerCorners, markerIds); if (markerIds.rows > 0) { // camera intrinsics (you should calibrate!) let fx = 700, fy = 700, cx = video.videoWidth / 2, cy = video.videoHeight / 2; const cameraMatrix = cv.matFromArray(3, 3, cv.CV_64F, [ fx, 0, cx, 0, fy, cy, 0, 0, 1 ]); const distCoeffs = cv.Mat.zeros(1, 5, cv.CV_64F); const rvec = new cv.Mat(); const tvec = new cv.Mat(); const markerLength = 0.1; // 100 mm = 0.1 m cv.aruco.estimatePoseSingleMarkers( markerCorners, markerLength, cameraMatrix, distCoeffs, rvec, tvec ); // Convert pose to Three.js matrix const rotation = new THREE.Matrix4(); const translation = new THREE.Matrix4(); // Rodrigues -> rotation matrix let R = new cv.Mat(); cv.Rodrigues(rvec.row(0), R); const r = R.data64F; rotation.set( r[0], r[1], r[2], 0, r[3], r[4], r[5], 0, r[6], r[7], r[8], 0, 0, 0, 0, 1 ); const t = tvec.data64F; translation.set( 1, 0, 0, t[0], 0, 1, 0, t[1], 0, 0, 1, t[2], 0, 0, 0, 1 ); // Marker in camera coordinates const markerInCamera = new THREE.Matrix4().multiplyMatrices(translation, rotation); // Invert to get camera→marker (world) const cameraToMarker = new THREE.Matrix4().copy(markerInCamera).invert(); markerGroup.matrixAutoUpdate = false; markerGroup.matrix.copy(cameraToMarker); markerGroup.visible = true; // Cleanup R.delete(); } else { markerGroup.visible = false; } src.delete(); gray.delete(); markerCorners.delete(); markerIds.delete(); }
